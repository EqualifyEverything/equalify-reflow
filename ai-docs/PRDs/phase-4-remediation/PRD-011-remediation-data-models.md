## Problem Statement

The remediation pipeline requires new data models to support the observation-first design pattern. Unlike the current extraction-only approach, the remediation pipeline generates observations (discrepancies between visual and markup), consolidates them into proposals (actionable search-replace edits), and tracks human review decisions.

These models must support:
- Per-page feature detection for agent routing
- Observation lifecycle (open → resolved/wont_fix/manual)
- Proposal workflow (pending → approved/rejected → applied/failed)
- Job substatus tracking within the "processing" state
- S3 storage schema for observations and proposals

```python
from datetime import datetime
from typing import Literal
from pydantic import BaseModel, Field


class PageFeatures(BaseModel):
    """Features detected on a single page by the Analysis agent."""

    page_num: int = Field(..., ge=1, description="1-indexed page number")

    # Content detection
    has_images: bool = Field(default=False)
    image_count: int = Field(default=0, ge=0)
    has_tables: bool = Field(default=False)
    table_count: int = Field(default=0, ge=0)
    has_lists: bool = Field(default=False)
    has_code_blocks: bool = Field(default=False)
    has_math: bool = Field(default=False)

    # Layout
    layout_type: Literal["single_column", "two_column", "mixed"] = "single_column"
    has_headers_footers: bool = Field(default=False)

    # Complexity assessment
    complexity_score: float = Field(default=0.5, ge=0.0, le=1.0)
    complexity_factors: list[str] = Field(default_factory=list)


class DocumentManifest(BaseModel):
    """Complete document analysis from the Analysis phase.

    This manifest is produced by the Analysis agent (Sonnet) and guides:
    - The Extraction agent (Haiku) for markdown generation
    - Agent routing decisions (which specialized agents to run)
    - Per-page context for specialized analysis
    """

    job_id: str = Field(..., description="Associated job ID")

    # Document metadata
    document_title: str = Field(default="Untitled")
    document_type: str = Field(default="unknown")  # syllabus, lecture, exam, etc.
    total_pages: int = Field(..., ge=1)

    # Structure (from existing HeadingTree)
    heading_tree_json: str = Field(..., description="Serialized HeadingTree")

    # Per-page analysis
    page_features: list[PageFeatures] = Field(default_factory=list)

    # Agent routing
    required_agents: list[str] = Field(default_factory=list)  # ["figures", "tables"]
    skip_agents: list[str] = Field(default_factory=list)

    # Confidence
    analysis_confidence: float = Field(default=0.8, ge=0.0, le=1.0)
    analysis_notes: str = Field(default="")

    # Metadata
    created_at: datetime = Field(default_factory=datetime.utcnow)
    analysis_model: str = Field(default="claude-sonnet-4-5")
```

#### Observation Model

```python
class ObservationLocation(BaseModel):
    """Location reference for an observation."""

    location_type: Literal["element", "range", "region"] = "region"
    value: str = Field(..., description="Selector, line range, or description")
    page_num: int = Field(..., ge=1)


class Observation(BaseModel):
    """A discrepancy between visual presentation and semantic markup.

    Observations are the atomic unit of the remediation pipeline.
    They can be generated by:
    - Analysis agent (initial document scan)
    - Specialized agents (figures, tables, structure, typography)
    - Human reviewers (manual observations)

    Observations do NOT map directly to WCAG criteria - that mapping
    can be derived later for compliance reporting.
    """

    id: str = Field(..., description="UUID for this observation")
    job_id: str = Field(..., description="Associated job ID")

    # Source
    agent: str = Field(..., description="Agent that created: analysis, figures, tables, structure, typography, human")
    source: Literal["agent", "human"] = "agent"

    # What the AI/human perceived
    visual_description: str = Field(
        ...,
        description="What is visually presented in the PDF"
    )
    markup_description: str = Field(
        ...,
        description="How it's currently represented in markup"
    )

    # Location
    location: ObservationLocation

    # Assessment
    confidence: float = Field(default=0.8, ge=0.0, le=1.0)
    severity: Literal["critical", "major", "minor"] = "major"

    # Routing
    route: Literal["auto", "manual"] = "auto"
    manual_reason: str | None = Field(
        default=None,
        description="Why this needs human judgment"
    )

    # Lifecycle
    status: Literal["open", "resolved", "wont_fix", "manual"] = "open"
    resolved_by: str | None = Field(
        default=None,
        description="Proposal ID that resolved this"
    )

    # Metadata
    created_at: datetime = Field(default_factory=datetime.utcnow)

    # Optional: Human edit context
    human_comment: str | None = Field(
        default=None,
        description="Comment from human when submitting observation"
    )
```

#### Proposal Model

```python
class SearchReplaceDiff(BaseModel):
    """A single search-replace edit operation."""

    search: str = Field(
        ...,
        min_length=1,
        description="Exact text to find in document"
    )
    replace: str = Field(
        ...,
        description="Text to substitute (can be empty for deletion)"
    )


class Proposal(BaseModel):
    """A suggested edit that resolves one or more observations.

    Proposals are the unit of human review. Each proposal:
    - Resolves one or more observations
    - Contains a search-replace diff
    - Includes justification for the change
    - Requires human approval before application
    """

    id: str = Field(..., description="UUID for this proposal")
    job_id: str = Field(..., description="Associated job ID")

    # What this addresses
    resolves: list[str] = Field(
        ...,
        min_length=1,
        description="Observation IDs this proposal resolves"
    )

    # The edit
    diff: SearchReplaceDiff

    # Justification (critical for human review)
    justification: str = Field(
        ...,
        description="Why these observations are grouped and how the edit addresses them"
    )

    # Context
    page_nums: list[int] = Field(
        default_factory=list,
        description="Pages affected by this proposal"
    )
    estimated_impact: str = Field(
        default="",
        description="Human-readable impact summary"
    )

    # Routing
    route: Literal["auto", "manual"] = "auto"

    # Lifecycle
    status: Literal["pending", "approved", "rejected", "applied", "failed"] = "pending"
    failure_reason: str | None = Field(default=None)

    # Review
    reviewed_by: str | None = Field(default=None)
    reviewed_at: datetime | None = Field(default=None)
    review_notes: str | None = Field(default=None)

    # Metadata
    created_at: datetime = Field(default_factory=datetime.utcnow)
```

```python
import json
from typing import Any

class RemediationStorageService:

    def __init__(self, storage_service: StorageService):

    async def save_manifest(self, job_id: str, manifest: DocumentManifest) -> str:
        """Save document manifest to S3."""
        content = manifest.model_dump_json(indent=2)
        key = f"{job_id}/manifest.json"
        await self.storage.upload_result(
            job_id=job_id,
            content=content,
            format="json",
            suffix="manifest"
        )
        return key

        """Save observations to S3."""
        content = json.dumps(
            [obs.model_dump() for obs in observations],
            indent=2,

        proposals: list[Proposal]

        """Save proposals to S3."""
        pass

    async def load_proposals(self, job_id: str) -> list[Proposal]:
        """Load proposals from S3."""
        pass

    async def update_proposal_status(
        self,
        job_id: str,
        proposal_id: str,
        status: str,
        **fields: Any
    ) -> None:
        """Update a single proposal's status."""
        proposals = await self.load_proposals(job_id)
        for p in proposals:
            if p.id == proposal_id:
                p.status = status
                for key, value in fields.items():
                    setattr(p, key, value)
                break
        await self.save_proposals(job_id, proposals)
```
